	# ROOT is the winner; FROM is the annexed country.
	on_annex = {
		effect = {
			if = {
				limit = {
					NOT = { has_country_flag = KR_checking_for_possible_annexations }
					is_subject = no
				}
				set_country_flag = KR_checking_for_possible_annexations
				country_event = { id = annex.1 days = 1 random_days = 3 }
			}
		}
	}

	# ROOT is the winner; FROM is the loser.
	on_peaceconference_ended = {
		effect = {
			if = {
				limit = {
					NOT = { has_country_flag = KR_checking_for_possible_annexations }
					is_subject = no
				}
				set_country_flag = KR_checking_for_possible_annexations
				country_event = { id = annex.1 days = 1 random_days = 3 }
			}
		}
	}

	# ROOT is the new controller; FROM.FROM is the state.
	on_state_control_changed = {
		effect = {
			if = {
				limit = {
					owns_state = FROM.FROM
					is_in_array = { global.annexation_relevant_states = FROM.FROM }
					NOT = { has_country_flag = KR_checking_for_possible_annexations }
					is_subject = no
				}
				set_country_flag = KR_checking_for_possible_annexations
				country_event = { id = annex.1 days = 1 random_days = 3 }
			}
		}
	}
