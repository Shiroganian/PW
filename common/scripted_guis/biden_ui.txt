# Minimal KX-style leader GUI implementation for GER.
# Gustav Noske is selected by GER_Gustav_Noske_HoG rather than a minister idea.
scripted_gui = {
	view_pm = {
		context_type = player_context
		parent_window_token = politics_tab
		window_name = "joe_biden_gui"

		ai_check = {
			always = no
		}

		triggers = {
			pm_frame_visible = {
				has_country_flag = HOG_veiw
			}

			leader_button_visible = {
				uses_leader_gui = yes
			}

			Gustav_Noske_visible = {
				AND = {
					has_country_flag = HOG_veiw
					has_country_flag = GER_Gustav_Noske_HoG
				}
			}

			pm_name_visible = {
				has_country_flag = HOG_veiw
			}
		}

		visible = {
			uses_leader_gui = yes
		}

		effects = {
			leader_button_click = {
				if = {
					limit = {
						has_country_flag = HOG_veiw
					}
					clr_country_flag = HOG_veiw
					else = {
						set_country_flag = HOG_veiw
					}
				}
			}
		}
	}
}
