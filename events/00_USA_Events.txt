						fascist = 4
						national_socialist = 3
					}
				}
			}
			else_if = {
				limit = { 
					has_completed_focus = USA_1936_Party_Thesis_Antithesis_Synthesis
				}
				set_popularities = {
					totalitarian_socialist = 2
					communist = 3
					libertarian_socialist = 3
					social_democrat = 35
					social_liberal = 15
					market_liberal = 10
					conservative = 10
					authoritarian_democrat = 15
					nationalist = 0
					fascist = 4
					national_socialist = 3
				}
			}
			else_if = {
				limit = { 
					has_completed_focus = USA_1936_Party_Meet_With_Byrd
				}
				set_popularities = {
					totalitarian_socialist = 2
					communist = 3
					libertarian_socialist = 3

					social_liberal = 10
					market_liberal = 10
					conservative = 10

				}
			}
			else = {
